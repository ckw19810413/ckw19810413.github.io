{{- /* Hreflang alternate links for multilingual SEO. Emits one per translated locale plus a self-referential link, and a single x-default that consistently points to the DEFAULT content language (zh-tw) version of *this* page from every translation — x-default must be the same URL across the whole translation set, never each page pointing at itself. */ -}} {{- if .IsTranslated }} {{- /* Resolve the default content language: with defaultContentLanguageInSubdir=false it is the only home served without a language subdir (RelPermalink "/"). */ -}} {{- $defaultLang := site.Language.Lang -}} {{- range site.Home.AllTranslations -}} {{- if eq .RelPermalink "/" -}}{{- $defaultLang = .Language.Lang -}}{{- end -}} {{- end -}} {{- /* This page's default-language permalink (falls back to self if untranslated there). */ -}} {{- $xDefault := .Permalink -}} {{- range .AllTranslations -}} {{- if eq .Language.Lang $defaultLang -}}{{- $xDefault = .Permalink -}}{{- end -}} {{- end -}} {{- range .Translations }} {{- end }} {{- else if .IsHome }} {{- /* Fallback for a home page that isn't translated: still advertise every locale, with x-default anchored to the default-language root. */ -}} {{- end }}
