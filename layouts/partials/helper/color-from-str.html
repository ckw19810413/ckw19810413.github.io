{{/* This helper is used to generate a background color and text color from a string Used to generate deterministic colors for tags, categories, etc. Override: uses string-length-based color to avoid hash function compatibility issues across different Hugo versions (0.157 → 0.164+) */}} {{- $str := . -}} {{- $h := mod (mul (len $str) 37) 360 -}} {{- $backgroundColor := printf "hsl(%d, 60%%, 84%%)" $h -}} {{- $textColor := printf "hsl(%d, 60%%, 15%%)" $h -}} {{- return dict "BackgroundColor" $backgroundColor "TextColor" $textColor "Hue" $h -}}
